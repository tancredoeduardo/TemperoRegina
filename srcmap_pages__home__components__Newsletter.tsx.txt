import { useState, FormEvent } from "react";
import { Send, Check, Mail, Sparkles } from "lucide-react";
import ScrollReveal from "@/components/base/ScrollReveal";

export default function Newsletter() {
  const [email, setEmail] = useState("");
  const [submitted, setSubmitted] = useState(false);
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (e: FormEvent) => {
    e.preventDefault();
    if (!email || !email.includes("@")) return;
    setLoading(true);

    const formData = new URLSearchParams();
    formData.append("email", email);

    try {
      await fetch("https://readdy.ai/api/form/d7qmr6ml0bai2p3hafag", {
        method: "POST",
        headers: { "Content-Type": "application/x-www-form-urlencoded" },
        body: formData.toString(),
      });
    } catch {
      // ignore
    }

    setLoading(false);
    setSubmitted(true);
    setEmail("");
    setTimeout(() => setSubmitted(false), 5000);
  };

  return (
    <section className="relative bg-regina-red py-14 md:py-20 overflow-hidden">
      {/* Decorative circles */}
      <div className="absolute -top-20 -right-20 w-64 h-64 rounded-full bg-white/5" />
      <div className="absolute -bottom-16 -left-16 w-48 h-48 rounded-full bg-white/5" />

      <div className="relative max-w-3xl mx-auto px-4 md:px-6 text-center">
        <ScrollReveal>
          <div className="inline-flex items-center gap-2 mb-3 bg-white/10 rounded-full px-3 py-1">
            <Sparkles className="w-3.5 h-3.5 text-regina-yellow" />
            <span className="text-xs font-medium text-white/90 uppercase tracking-wider">Newsletter</span>
          </div>
          <h2 className="text-xl md:text-3xl font-bold text-white uppercase tracking-wide mb-3">
            Fique por Dentro das Novidades
          </h2>
          <p className="text-sm text-white/80 max-w-md mx-auto mb-8">
            Receba em primeira mão lançamentos, receitas exclusivas, promoções e eventos da Tempero Regina.
          </p>
        </ScrollReveal>

        <ScrollReveal delay={0.15}>
          <form
            id="newsletter-form"
            data-readdy-form
            onSubmit={handleSubmit}
            className="flex flex-col sm:flex-row items-stretch gap-3 max-w-lg mx-auto"
          >
            <div className="relative flex-1">
              <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
              <input
                type="email"
                name="email"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                placeholder="Digite seu melhor e-mail"
                required
                className="w-full pl-10 pr-4 py-3 rounded-md bg-white text-sm text-gray-900 placeholder-gray-400 focus:outline-none focus:ring-2 focus:ring-regina-yellow/50 border-0"
              />
            </div>
            <button
              type="submit"
              disabled={loading || submitted}
              className="bg-regina-yellow text-regina-dark text-sm font-semibold px-6 py-3 rounded-md hover:bg-regina-yellow-light transition-all duration-300 whitespace-nowrap flex items-center justify-center gap-2 disabled:opacity-70"
            >
              {submitted ? (
                <>
                  <Check className="w-4 h-4" />
                  Inscrito!
                </>
              ) : loading ? (
                <span className="w-4 h-4 border-2 border-regina-dark/30 border-t-regina-dark rounded-full animate-spin" />
              ) : (
                <>
                  <Send className="w-4 h-4" />
                  INSCREVER-SE
                </>
              )}
            </button>
          </form>
          {submitted && (
            <p className="mt-3 text-xs text-white/80 animate-fade-in">
              Obrigado! Você receberá nossas novidades em breve.
            </p>
          )}
          <p className="mt-3 text-[10px] text-white/50">
            Ao se inscrever, você concorda em receber e-mails da Tempero Regina. Pode cancelar a qualquer momento.
          </p>
        </ScrollReveal>
      </div>
    </section>
  );
}
