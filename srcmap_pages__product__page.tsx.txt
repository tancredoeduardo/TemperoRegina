import { useParams, Link } from "react-router-dom";
import { useState } from "react";
import { Download, ArrowLeft, Share2, ChevronRight } from "lucide-react";
import Navbar from "@/components/feature/Navbar";
import Footer from "@/components/feature/Footer";
import ScrollReveal from "@/components/base/ScrollReveal";
import { findProductById, getAllProducts, productCategories } from "@/mocks/products";

export default function ProductDetail() {
  const { id } = useParams<{ id: string }>();
  const [showToast, setShowToast] = useState(false);
  const result = id ? findProductById(id) : null;

  const allProducts = getAllProducts();
  const relatedProducts = allProducts
    .filter((p) => p.id !== id && p.categoryId === result?.category.id)
    .slice(0, 4);

  if (!result) {
    return (
      <div className="min-h-screen bg-regina-gray-warm flex flex-col">
        <Navbar />
        <div className="flex-1 flex items-center justify-center px-4">
          <div className="text-center">
            <h1 className="text-2xl md:text-3xl font-bold text-gray-800 mb-4">Produto não encontrado</h1>
            <p className="text-gray-600 mb-6">O produto que você procura não está disponível.</p>
            <Link
              to="/"
              className="inline-flex items-center gap-2 bg-regina-red text-white px-5 py-2.5 rounded-md text-sm font-medium hover:bg-regina-red-dark transition-all duration-300 hover:shadow-lg"
            >
              <ArrowLeft className="w-4 h-4" />
              Voltar para Home
            </Link>
          </div>
        </div>
        <Footer />
      </div>
    );
  }

  const { product, category } = result;

  const handleShare = () => {
    if (navigator.share) {
      navigator
        .share({
          title: product.name,
          text: `Confira o ${product.name} da Tempero Regina!`,
          url: window.location.href,
        })
        .catch(() => {});
    } else {
      navigator.clipboard.writeText(window.location.href);
      setShowToast(true);
      setTimeout(() => setShowToast(false), 2000);
    }
  };

  return (
    <div className="min-h-screen bg-regina-gray-warm flex flex-col">
      <Navbar />

      {/* Breadcrumb */}
      <div className="pt-20 md:pt-24 bg-white border-b border-gray-100">
        <div className="max-w-6xl mx-auto px-4 md:px-6 py-3">
          <nav className="flex items-center gap-2 text-xs md:text-sm text-gray-500">
            <Link to="/" className="hover:text-regina-red transition-colors">Home</Link>
            <ChevronRight className="w-3 h-3" />
            <span className="text-gray-800 font-medium">{category.title}</span>
            <ChevronRight className="w-3 h-3" />
            <span className="text-regina-red font-medium truncate max-w-[200px] md:max-w-sm">{product.name}</span>
          </nav>
        </div>
      </div>

      {/* Product Detail */}
      <ScrollReveal>
      <section className="flex-1 py-8 md:py-14">
        <div className="max-w-6xl mx-auto px-4 md:px-6">
          <div className="flex flex-col lg:flex-row gap-8 md:gap-12">
            {/* Image */}
            <div className="w-full lg:w-1/2 shrink-0">
              <div className="bg-white rounded-2xl p-6 md:p-10 border border-gray-100 hover:shadow-xl transition-all duration-300 group">
                <div className="aspect-square bg-gradient-to-br from-gray-50 to-gray-100 rounded-xl flex items-center justify-center p-4 group-hover:from-regina-yellow/5 group-hover:to-regina-red/5 transition-colors duration-300">
                  <img
                    src={product.image}
                    alt={product.name}
                    className="w-full h-full object-contain group-hover:scale-105 transition-transform duration-500"
                  />
                </div>
              </div>
            </div>

            {/* Info */}
            <div className="flex-1">
              <span className="inline-block bg-regina-red/10 text-regina-red text-xs font-bold px-3 py-1 rounded-full mb-3 uppercase tracking-wide">
                {category.title}
              </span>
              <h1 className="text-2xl md:text-4xl font-bold text-gray-900 mb-4 leading-tight">
                {product.name}
              </h1>
              <p className="text-sm md:text-base text-gray-600 leading-relaxed mb-6">
                Produto da linha <strong className="text-regina-red">{category.title}</strong> da Tempero Regina.
                Qualidade e tradição desde 1967, produzido com os melhores ingredientes para
                deixar suas receitas ainda mais saborosas. O sabor que conquista o Brasil.
              </p>

              <div className="space-y-3 mb-8">
                <div className="flex items-center gap-3 text-sm text-gray-700">
                  <div className="w-2 h-2 rounded-full bg-regina-red" />
                  <span>Produto 100% natural</span>
                </div>
                <div className="flex items-center gap-3 text-sm text-gray-700">
                  <div className="w-2 h-2 rounded-full bg-regina-yellow" />
                  <span>Qualidade garantida desde 1967</span>
                </div>
                <div className="flex items-center gap-3 text-sm text-gray-700">
                  <div className="w-2 h-2 rounded-full bg-regina-red" />
                  <span>Disponível em supermercados de todo o Nordeste</span>
                </div>
              </div>

              <div className="flex flex-col sm:flex-row gap-3">
                <a
                  href={product.downloadUrl}
                  download
                  className="group inline-flex items-center justify-center gap-2 bg-regina-red text-white text-sm font-medium px-6 py-3 rounded-md hover:bg-regina-red-dark transition-all duration-300 hover:shadow-lg hover:-translate-y-0.5"
                >
                  <Download className="w-4 h-4 group-hover:animate-bounce" />
                  Download da Imagem
                </a>
                <button
                  type="button"
                  onClick={handleShare}
                  className="inline-flex items-center justify-center gap-2 bg-white text-gray-700 border border-gray-200 text-sm font-medium px-6 py-3 rounded-md hover:border-regina-red hover:text-regina-red transition-all duration-300"
                >
                  <Share2 className="w-4 h-4" />
                  Compartilhar
                </button>
              </div>

              {/* Share toast */}
              {showToast && (
                <div className="mt-3 p-2 bg-green-50 border border-green-200 rounded-md text-green-700 text-sm animate-fade-in">
                  Link copiado para a área de transferência!
                </div>
              )}
            </div>
          </div>
        </div>
      </section>
      </ScrollReveal>

      <ScrollReveal>
      {/* Related Products */}
      {relatedProducts.length > 0 && (
        <section className="py-8 md:py-14 bg-white border-t border-gray-100">
          <div className="max-w-6xl mx-auto px-4 md:px-6">
            <h2 className="text-xl md:text-2xl font-bold text-gray-900 mb-6 md:mb-8 uppercase tracking-wide">
              Produtos Relacionados
            </h2>
            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-4 md:gap-5">
              {relatedProducts.map((p) => (
                <Link
                  key={p.id}
                  to={`/produto/${p.id}`}
                  className="group bg-regina-gray-warm rounded-xl border border-gray-100 overflow-hidden transition-all duration-300 hover:shadow-lg hover:-translate-y-1 hover:border-regina-red/20"
                >
                  <div className="aspect-square bg-white flex items-center justify-center p-3">
                    <img
                      src={p.image}
                      alt={p.name}
                      className="w-full h-full object-contain group-hover:scale-105 transition-transform duration-300"
                    />
                  </div>
                  <div className="p-3">
                    <h4 className="text-xs md:text-sm text-gray-800 leading-tight line-clamp-2 group-hover:text-regina-red transition-colors duration-300">
                      {p.name}
                    </h4>
                  </div>
                </Link>
              ))}
            </div>
          </div>
        </section>
      )}

      {/* Other Categories */}
      <section className="py-8 md:py-14 bg-regina-gray-warm">
        <div className="max-w-6xl mx-auto px-4 md:px-6">
          <h2 className="text-xl md:text-2xl font-bold text-gray-900 mb-6 md:mb-8 uppercase tracking-wide">
            Outras Categorias
          </h2>
          <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-6 gap-3 md:gap-4">
            {productCategories
              .filter((c) => c.id !== category.id)
              .map((cat) => (
                <Link
                  key={cat.id}
                  to="/"
                  className="group flex flex-col items-center justify-center bg-white rounded-xl p-4 md:p-5 border border-gray-100 hover:border-regina-red/20 hover:shadow-lg hover:-translate-y-1 transition-all duration-300 cursor-pointer"
                >
                  {cat.products[0] && (
                    <img
                      src={cat.products[0].image}
                      alt={cat.title}
                      className="w-12 h-12 md:w-16 md:h-16 object-contain mb-2 group-hover:scale-110 transition-transform duration-300"
                    />
                  )}
                  <span className="text-xs md:text-sm font-medium text-gray-700 text-center group-hover:text-regina-red transition-colors duration-300">
                    {cat.title}
                  </span>
                </Link>
              ))}
          </div>
        </div>
      </section>
      </ScrollReveal>

      <Footer />
    </div>
  );
}
