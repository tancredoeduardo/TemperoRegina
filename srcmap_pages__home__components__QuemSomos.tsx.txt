import { useEffect, useRef, useState } from "react";
import ScrollReveal from "@/components/base/ScrollReveal";

export default function QuemSomos() {
  const sectionRef = useRef<HTMLElement>(null);
  const parallaxRef = useRef<HTMLDivElement>(null);
  const [parallaxY, setParallaxY] = useState(0);

  useEffect(() => {
    const handleScroll = () => {
      if (!sectionRef.current) return;
      const rect = sectionRef.current.getBoundingClientRect();
      const scrollProgress = -rect.top / (window.innerHeight + rect.height);
      setParallaxY(scrollProgress * 40);
    };
    window.addEventListener("scroll", handleScroll, { passive: true });
    return () => window.removeEventListener("scroll", handleScroll);
  }, []);

  return (
    <section
      id="quemsomos"
      ref={sectionRef}
      className="relative bg-regina-red py-12 md:py-20 overflow-hidden"
    >
      {/* Parallax decorative background */}
      <div
        ref={parallaxRef}
        className="absolute inset-0 pointer-events-none opacity-10"
        style={{ transform: `translateY(${parallaxY}px)` }}
      >
        <div className="absolute -top-20 -right-20 w-80 h-80 rounded-full bg-white/20" />
        <div className="absolute -bottom-32 -left-32 w-96 h-96 rounded-full bg-white/10" />
      </div>

      <div className="relative max-w-6xl mx-auto px-4 md:px-6">
        <div className="flex flex-col md:flex-row items-start gap-8 md:gap-12">
          <ScrollReveal className="flex-1">
            <h2 className="text-2xl md:text-4xl font-bold text-white mb-6 uppercase tracking-wide">
              Quem Somos
            </h2>
            <div className="text-white/90 text-sm md:text-base leading-relaxed space-y-4">
              <p className="hover:text-white transition-colors duration-300">
                A Indústria de Tempero Regina Ltda nasceu em 1967, em Mossoró-RN. Um negócio de família que agradou o paladar dos mossoroenses, cresceu e hoje dá mais sabor à vida de milhares de brasileiros, se tornando a maior indústria de temperos e condimentos do Nordeste. Regina sabe que nada cai do céu, nem vem de mão beijada, além do sabor, aqui tem batalha, perseverança, suor, otimismo e muita criatividade. Cremos num mundo melhor, mais gostoso e trabalhamos para isso, sempre com muita humildade.
              </p>
              <p className="hover:text-white transition-colors duration-300">
                Por isso, o que começou com a venda de um único Tempero Caseiro, evoluiu para uma linha com mais de 100 produtos alimentícios. As primeiras entregas, feitas de bicicleta pelos bairros mossoroenses, trilharam um grande caminho até chegar a uma frota com 18 caminhões, que atualmente percorrem o Nordeste e partes do Norte do Brasil.
              </p>
            </div>
          </ScrollReveal>
          <ScrollReveal delay={0.2} className="hidden md:block w-32 shrink-0">
            <img
              src="https://temperoregina.com.br/lib/img/quemsomos-logo.png"
              alt="Logo Regina"
              className="w-full h-auto object-contain hover:scale-110 transition-transform duration-500"
            />
          </ScrollReveal>
        </div>
      </div>
    </section>
  );
}
