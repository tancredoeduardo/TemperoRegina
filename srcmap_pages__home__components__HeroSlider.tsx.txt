import { useState, useEffect, useCallback } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { ChevronLeft, ChevronRight } from "lucide-react";
import { heroSlides } from "@/mocks/home";

const slideVariants = {
  enter: (direction: number) => ({
    x: direction > 0 ? "100%" : "-100%",
    opacity: 0,
    scale: 1.05,
  }),
  center: {
    x: 0,
    opacity: 1,
    scale: 1,
    transition: { duration: 0.7, ease: [0.32, 0.72, 0, 1] },
  },
  exit: (direction: number) => ({
    x: direction > 0 ? "-100%" : "100%",
    opacity: 0,
    scale: 0.98,
    transition: { duration: 0.5, ease: [0.32, 0.72, 0, 1] },
  }),
};

const textContainerVariants = {
  hidden: { opacity: 0 },
  visible: {
    opacity: 1,
    transition: {
      staggerChildren: 0.12,
      delayChildren: 0.3,
    },
  },
};

const textItemVariants = {
  hidden: { opacity: 0, y: 30 },
  visible: {
    opacity: 1,
    y: 0,
    transition: { duration: 0.6, ease: [0.25, 0.46, 0.45, 0.94] },
  },
};

export default function HeroSlider() {
  const [current, setCurrent] = useState(0);
  const [direction, setDirection] = useState(0);

  const paginate = useCallback((newDirection: number) => {
    setDirection(newDirection);
    setCurrent((prev) => {
      const next = prev + newDirection;
      if (next < 0) return heroSlides.length - 1;
      if (next >= heroSlides.length) return 0;
      return next;
    });
  }, []);

  const goTo = useCallback((index: number) => {
    setDirection(index > current ? 1 : -1);
    setCurrent(index);
  }, [current]);

  useEffect(() => {
    const timer = setInterval(() => paginate(1), 6000);
    return () => clearInterval(timer);
  }, [paginate]);

  return (
    <section id="hero" className="relative w-full h-[380px] md:h-[540px] lg:h-[620px] overflow-hidden bg-gray-900">
      <AnimatePresence initial={false} custom={direction} mode="popLayout">
        <motion.div
          key={current}
          custom={direction}
          variants={slideVariants}
          initial="enter"
          animate="center"
          exit="exit"
          className="absolute inset-0"
        >
          <img
            src={heroSlides[current].image}
            alt={heroSlides[current].alt}
            className="w-full h-full object-cover object-top"
          />
          <div className="absolute inset-0 bg-gradient-to-b from-black/40 via-black/30 to-black/60" />
        </motion.div>
      </AnimatePresence>

      {/* Texto com stagger */}
      <div className="absolute inset-0 z-20 flex items-center justify-center">
        <div className="max-w-6xl mx-auto px-4 md:px-8 w-full text-center">
          <AnimatePresence mode="wait">
            <motion.div
              key={`text-${current}`}
              variants={textContainerVariants}
              initial="hidden"
              animate="visible"
              exit={{ opacity: 0, transition: { duration: 0.3 } }}
            >
              <motion.p
                variants={textItemVariants}
                className="text-xs md:text-sm font-semibold text-regina-yellow uppercase tracking-[0.2em] mb-3 md:mb-4"
              >
                Desde 1967
              </motion.p>
              <motion.h1
                variants={textItemVariants}
                className="text-3xl md:text-5xl lg:text-6xl font-bold text-white uppercase tracking-wide mb-3 md:mb-4 drop-shadow-2xl"
              >
                Tempero Regina
              </motion.h1>
              <motion.p
                variants={textItemVariants}
                className="text-sm md:text-lg text-white/85 max-w-xl mx-auto drop-shadow-lg"
              >
                O sabor que conquista o Brasil há mais de 57 anos
              </motion.p>
              <motion.div variants={textItemVariants} className="mt-6 md:mt-8">
                <a
                  href="/catalogo"
                  className="inline-flex items-center gap-2 bg-regina-red text-white text-sm font-semibold px-6 py-3 rounded-md hover:bg-regina-red-dark transition-all duration-300 whitespace-nowrap hover:shadow-xl hover:-translate-y-0.5"
                >
                  Conheça Nossos Produtos
                  <ChevronRight className="w-4 h-4" />
                </a>
              </motion.div>
            </motion.div>
          </AnimatePresence>
        </div>
      </div>

      {/* Navigation Arrows */}
      <button
        type="button"
        onClick={() => paginate(-1)}
        className="absolute left-3 md:left-8 top-1/2 -translate-y-1/2 z-20 w-10 h-10 md:w-12 md:h-12 flex items-center justify-center bg-white/15 hover:bg-white/30 backdrop-blur-sm rounded-full text-white transition-all duration-300 border border-white/20"
        aria-label="Anterior"
      >
        <ChevronLeft className="w-5 h-5" />
      </button>
      <button
        type="button"
        onClick={() => paginate(1)}
        className="absolute right-3 md:right-8 top-1/2 -translate-y-1/2 z-20 w-10 h-10 md:w-12 md:h-12 flex items-center justify-center bg-white/15 hover:bg-white/30 backdrop-blur-sm rounded-full text-white transition-all duration-300 border border-white/20"
        aria-label="Próximo"
      >
        <ChevronRight className="w-5 h-5" />
      </button>

      {/* Progress Dots */}
      <div className="absolute bottom-6 md:bottom-10 left-1/2 -translate-x-1/2 z-20 flex items-center gap-2">
        {heroSlides.map((_, index) => (
          <button
            key={index}
            type="button"
            onClick={() => goTo(index)}
            className="group relative h-2 rounded-full overflow-hidden bg-white/30 transition-all duration-500"
            style={{ width: index === current ? 32 : 8 }}
            aria-label={`Slide ${index + 1}`}
          >
            {index === current && (
              <motion.div
                className="absolute inset-0 bg-regina-yellow"
                initial={{ scaleX: 0, originX: 0 }}
                animate={{ scaleX: 1 }}
                transition={{ duration: 6, ease: "linear" }}
                key={`progress-${current}`}
              />
            )}
          </button>
        ))}
      </div>
    </section>
  );
}
