import { useState, useEffect, useRef } from "react";
import { Link } from "react-router-dom";
import {
  Search,
  Clock,
  Users,
  X,
  ChefHat,
  Filter,
  Flame,
  ShoppingBag,
} from "lucide-react";
import Navbar from "@/components/feature/Navbar";
import Footer from "@/components/feature/Footer";
import { recipesData, categories } from "@/mocks/recipes";
import { motion, AnimatePresence } from "framer-motion";

/* ═══════════════════════════════
   Skeleton Card
   ═══════════════════════════════ */
function RecipeSkeleton() {
  return (
    <div className="bg-white rounded-xl overflow-hidden border border-gray-100 animate-pulse">
      <div className="h-44 md:h-52 bg-gray-200" />
      <div className="p-4 md:p-5 space-y-3">
        <div className="h-4 bg-gray-200 rounded w-3/4" />
        <div className="h-3 bg-gray-200 rounded w-1/2" />
      </div>
    </div>
  );
}

export default function RecipesPage() {
  const [activeCategory, setActiveCategory] = useState("Todas");
  const [searchTerm, setSearchTerm] = useState("");
  const [selectedRecipe, setSelectedRecipe] = useState<number | null>(null);
  const [heroVisible, setHeroVisible] = useState(false);
  const modalRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    setHeroVisible(true);
  }, []);

  useEffect(() => {
    if (selectedRecipe !== null) {
      document.body.style.overflow = "hidden";
    } else {
      document.body.style.overflow = "";
    }
    return () => {
      document.body.style.overflow = "";
    };
  }, [selectedRecipe]);

  const filtered = recipesData.filter((recipe) => {
    const matchesCategory =
      activeCategory === "Todas" || recipe.category === activeCategory;
    const matchesSearch =
      recipe.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
      recipe.ingredientes.some((ing) =>
        ing.toLowerCase().includes(searchTerm.toLowerCase())
      );
    return matchesCategory && matchesSearch;
  });

  const activeRecipe = selectedRecipe
    ? recipesData.find((r) => r.id === selectedRecipe)
    : null;

  return (
    <div className="min-h-screen bg-regina-gray-warm flex flex-col">
      <Navbar />

      {/* Hero */}
      <div
        className="relative w-full h-[260px] md:h-[360px] overflow-hidden"
        style={{ marginTop: "80px" }}
      >
        <img
          src="https://readdy.ai/api/search-image?query=A beautiful Brazilian kitchen scene with fresh colorful ingredients and spice jars on a wooden counter, chef hands preparing a delicious traditional meal with vibrant red and yellow seasonings, warm golden lighting, professional food photography, appetizing and inviting atmosphere, clean modern rustic kitchen background"
          alt="Receitas Tempero Regina"
          className="w-full h-full object-cover"
        />
        <div className="absolute inset-0 bg-gradient-to-b from-black/40 via-black/30 to-black/50" />
        <div className="absolute inset-0 flex items-center justify-center">
          <div className="text-center px-4">
            <h1
              className={`text-3xl md:text-5xl font-bold text-white uppercase tracking-wide mb-3 drop-shadow-lg transition-all duration-700 ${
                heroVisible ? "opacity-100 translate-y-0" : "opacity-0 translate-y-6"
              }`}
            >
              Receitas Regina
            </h1>
            <p
              className={`text-sm md:text-lg text-white/90 max-w-xl mx-auto drop-shadow-md transition-all duration-700 delay-150 ${
                heroVisible ? "opacity-100 translate-y-0" : "opacity-0 translate-y-6"
              }`}
            >
              As coisas mais simples da vida são sempre as mais gostosas. Basta temperar!
            </p>
          </div>
        </div>
      </div>

      {/* Filters & Search */}
      <section className="py-5 md:py-6 bg-white border-b border-gray-100 sticky top-16 md:top-20 z-30">
        <div className="max-w-6xl mx-auto px-4 md:px-6">
          <div className="flex flex-col md:flex-row gap-4 md:gap-6 items-start md:items-center">
            <div className="relative w-full md:w-72">
              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
              <input
                type="text"
                placeholder="Buscar receitas ou ingredientes..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="w-full pl-9 pr-4 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:border-regina-red bg-white"
              />
            </div>
            <div className="flex items-center gap-2 overflow-x-auto scrollbar-hide w-full md:w-auto">
              <Filter className="w-4 h-4 text-gray-400 shrink-0 hidden md:block" />
              {categories.map((cat) => (
                <button
                  key={cat}
                  type="button"
                  onClick={() => setActiveCategory(cat)}
                  className={`px-3 py-1.5 text-xs md:text-sm font-medium rounded-full whitespace-nowrap transition-all ${
                    activeCategory === cat
                      ? "bg-regina-red text-white"
                      : "bg-gray-100 text-gray-600 hover:bg-gray-200"
                  }`}
                >
                  {cat}
                </button>
              ))}
            </div>
          </div>
        </div>
      </section>

      {/* Recipes Grid */}
      <section className="flex-1 py-8 md:py-12">
        <div className="max-w-6xl mx-auto px-4 md:px-6">
          {filtered.length === 0 ? (
            <motion.div
              className="text-center py-12"
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
            >
              <ChefHat className="w-12 h-12 text-gray-300 mx-auto mb-3" />
              <p className="text-gray-500 text-sm md:text-base">
                Nenhuma receita encontrada.
              </p>
              <p className="text-gray-400 text-xs md:text-sm mt-1">
                Tente buscar por outro termo ou categoria.
              </p>
            </motion.div>
          ) : (
            <motion.div
              className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-5 md:gap-6"
              initial="hidden"
              animate="visible"
              variants={{
                hidden: {},
                visible: { transition: { staggerChildren: 0.06 } },
              }}
            >
              {filtered.map((recipe) => (
                <motion.div
                  key={recipe.id}
                  variants={{
                    hidden: { opacity: 0, y: 24, scale: 0.96 },
                    visible: { opacity: 1, y: 0, scale: 1 },
                  }}
                  transition={{ duration: 0.45, ease: [0.25, 0.46, 0.45, 0.94] }}
                  whileHover={{ y: -6, transition: { duration: 0.25 } }}
                  className="group bg-white rounded-xl overflow-hidden border border-gray-100 hover:shadow-xl hover:border-regina-red/15 transition-shadow duration-300 cursor-pointer"
                  onClick={() => setSelectedRecipe(recipe.id)}
                >
                  <div className="relative h-44 md:h-52 overflow-hidden">
                    <img
                      src={recipe.image}
                      alt={recipe.title}
                      className="w-full h-full object-cover group-hover:scale-110 transition-transform duration-700"
                      loading="lazy"
                    />
                    <div className="absolute inset-0 bg-gradient-to-t from-black/50 to-transparent" />
                    <span className="absolute bottom-3 left-3 bg-regina-yellow text-regina-dark text-[10px] md:text-xs font-bold px-2 py-1 rounded-full uppercase tracking-wide">
                      {recipe.category}
                    </span>
                    <motion.span
                      className={`absolute top-3 right-3 text-[10px] md:text-xs font-bold px-2 py-0.5 rounded-full ${
                        recipe.difficulty === "Fácil"
                          ? "bg-green-100 text-green-700"
                          : recipe.difficulty === "Médio"
                          ? "bg-yellow-100 text-yellow-700"
                          : "bg-red-100 text-red-700"
                      }`}
                      whileHover={{ scale: 1.08 }}
                    >
                      {recipe.difficulty}
                    </motion.span>
                  </div>
                  <div className="p-4 md:p-5">
                    <h3 className="text-sm md:text-base font-bold text-gray-900 mb-2 line-clamp-1 group-hover:text-regina-red transition-colors duration-300">
                      {recipe.title}
                    </h3>
                    <div className="flex items-center gap-4 text-xs text-gray-500">
                      <div className="flex items-center gap-1">
                        <Clock className="w-3.5 h-3.5" />
                        <span>{recipe.time}</span>
                      </div>
                      <div className="flex items-center gap-1">
                        <Users className="w-3.5 h-3.5" />
                        <span>{recipe.servings} porções</span>
                      </div>
                      <div className="flex items-center gap-1">
                        <Flame className="w-3.5 h-3.5" />
                        <span>{recipe.ingredientes.length} ingr.</span>
                      </div>
                    </div>
                  </div>
                </motion.div>
              ))}
            </motion.div>
          )}
        </div>
      </section>

      {/* CTA */}
      <motion.section
        className="py-10 md:py-14 bg-regina-red"
        initial={{ opacity: 0 }}
        whileInView={{ opacity: 1 }}
        viewport={{ once: true }}
      >
        <div className="max-w-4xl mx-auto px-4 md:px-6 text-center">
          <h2 className="text-xl md:text-2xl font-bold text-white uppercase tracking-wide mb-3">
            Tem uma receita com Regina?
          </h2>
          <p className="text-sm md:text-base text-white/80 mb-6 max-w-lg mx-auto">
            Compartilhe sua criação conosco e ela pode aparecer aqui!
          </p>
          <motion.a
            href="mailto:comercial@temperoregina.com.br?subject=Minha Receita com Tempero Regina"
            className="inline-flex items-center gap-2 bg-regina-yellow text-regina-dark text-sm font-bold px-6 py-3 rounded-md hover:bg-regina-yellow-light transition-all duration-300 whitespace-nowrap"
            whileHover={{ y: -3, boxShadow: "0 8px 24px rgba(0,0,0,0.15)" }}
            whileTap={{ scale: 0.97 }}
          >
            <ChefHat className="w-4 h-4" />
            Enviar Receita
          </motion.a>
        </div>
      </motion.section>

      {/* Recipe Modal */}
      <AnimatePresence>
        {activeRecipe && (
          <motion.div
            className="fixed inset-0 z-[60] bg-black/50 backdrop-blur-sm flex items-center justify-center px-4 py-8"
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            onClick={() => setSelectedRecipe(null)}
          >
            <motion.div
              ref={modalRef}
              className="bg-white rounded-2xl max-w-3xl w-full max-h-[90vh] overflow-y-auto"
              initial={{ opacity: 0, scale: 0.9, y: 30 }}
              animate={{ opacity: 1, scale: 1, y: 0 }}
              exit={{ opacity: 0, scale: 0.9, y: 30 }}
              transition={{ duration: 0.35, ease: [0.25, 0.46, 0.45, 0.94] }}
              onClick={(e) => e.stopPropagation()}
            >
              {/* Modal Header Image */}
              <div className="relative h-48 md:h-64 overflow-hidden rounded-t-2xl">
                <img
                  src={activeRecipe.image}
                  alt={activeRecipe.title}
                  className="w-full h-full object-cover"
                />
                <div className="absolute inset-0 bg-gradient-to-t from-black/60 to-transparent" />
                <motion.button
                  type="button"
                  onClick={() => setSelectedRecipe(null)}
                  className="absolute top-4 right-4 w-9 h-9 flex items-center justify-center rounded-full bg-white/90 hover:bg-white text-gray-700 transition-all"
                  aria-label="Fechar"
                  whileHover={{ scale: 1.1, rotate: 90 }}
                  whileTap={{ scale: 0.9 }}
                >
                  <X className="w-4 h-4" />
                </motion.button>
                <div className="absolute bottom-4 left-4 right-4">
                  <span className="inline-block bg-regina-yellow text-regina-dark text-xs font-bold px-2 py-1 rounded-full mb-2">
                    {activeRecipe.category}
                  </span>
                  <h2 className="text-xl md:text-2xl font-bold text-white drop-shadow-lg">
                    {activeRecipe.title}
                  </h2>
                </div>
              </div>

              {/* Modal Content */}
              <div className="p-5 md:p-8">
                {/* Meta */}
                <div className="flex flex-wrap items-center gap-4 mb-6 pb-6 border-b border-gray-100">
                  {[
                    { icon: <Clock className="w-4 h-4" />, text: activeRecipe.time },
                    { icon: <Users className="w-4 h-4" />, text: `${activeRecipe.servings} porções` },
                    { icon: <ChefHat className="w-4 h-4" />, text: activeRecipe.difficulty },
                  ].map((meta, i) => (
                    <motion.div
                      key={i}
                      className="flex items-center gap-1.5 text-sm text-gray-600"
                      initial={{ opacity: 0, y: 8 }}
                      animate={{ opacity: 1, y: 0 }}
                      transition={{ delay: 0.1 + i * 0.05 }}
                    >
                      <span className="text-regina-red">{meta.icon}</span>
                      <span>{meta.text}</span>
                    </motion.div>
                  ))}
                </div>

                <div className="flex flex-col md:flex-row gap-6 md:gap-10">
                  {/* Ingredientes */}
                  <div className="md:w-1/3 shrink-0">
                    <h3 className="text-sm font-bold text-gray-900 uppercase tracking-wide mb-3 flex items-center gap-2">
                      <ShoppingBag className="w-4 h-4 text-regina-red" />
                      Ingredientes
                    </h3>
                    <ul className="space-y-2">
                      {activeRecipe.ingredientes.map((ing, idx) => (
                        <motion.li
                          key={idx}
                          className="text-sm text-gray-700 leading-relaxed flex items-start gap-2"
                          initial={{ opacity: 0, x: -10 }}
                          animate={{ opacity: 1, x: 0 }}
                          transition={{ delay: 0.15 + idx * 0.03 }}
                        >
                          <span className="w-1.5 h-1.5 rounded-full bg-regina-red shrink-0 mt-1.5" />
                          <span>{ing}</span>
                        </motion.li>
                      ))}
                    </ul>
                  </div>

                  {/* Modo de Preparo */}
                  <div className="flex-1">
                    <h3 className="text-sm font-bold text-gray-900 uppercase tracking-wide mb-3 flex items-center gap-2">
                      <Flame className="w-4 h-4 text-regina-red" />
                      Modo de Preparo
                    </h3>
                    <ol className="space-y-3">
                      {activeRecipe.modoPreparo?.map((step, idx) => (
                        <motion.li
                          key={idx}
                          className="text-sm text-gray-700 leading-relaxed flex gap-3"
                          initial={{ opacity: 0, y: 8 }}
                          animate={{ opacity: 1, y: 0 }}
                          transition={{ delay: 0.2 + idx * 0.05 }}
                        >
                          <span className="w-6 h-6 shrink-0 rounded-full bg-regina-red/10 text-regina-red text-xs font-bold flex items-center justify-center">
                            {idx + 1}
                          </span>
                          <span>{step}</span>
                        </motion.li>
                      ))}
                    </ol>
                  </div>
                </div>
              </div>
            </motion.div>
          </motion.div>
        )}
      </AnimatePresence>

      <Footer />
    </div>
  );
}
