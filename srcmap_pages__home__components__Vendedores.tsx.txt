                key={r}
                type="button"
                onClick={() => setActiveRegiao(activeRegiao === r ? null : r)}

                  activeRegiao === r
                    ? "bg-regina-red text-white shadow-md"
                    : "bg-gray-100 text-gray-600 hover:bg-gray-200"
                }`}
              >
                {r}
              </button>
            ))}
          </div>
        </ScrollReveal>

        {/* Mapa SVG do Brasil */}
        <ScrollReveal delay={0.15}>
          <div className="relative w-full max-w-2xl mx-auto mb-8 select-none">
            <svg
              viewBox="0 0 320 440"

                fill="#FFD000"
                fillOpacity="0.15"
                stroke="none"
              />
              {/* Nordeste */}
              <path

                const isSelected = selectedSigla === estado.sigla;

                if (isFiltered) {

                      cy={pos.cy}

                      fill="#e5e5e5"
                      opacity="0.5"
                    />
                  );
                }

                return (
                  <g
                    key={estado.sigla}
                    className="cursor-pointer"
                    onClick={() => handleEstadoClick(estado.sigla)}
                    onMouseEnter={() => setHoveredEstado(estado.sigla)}
                    onMouseLeave={() => setHoveredEstado(null)}
                  >
                    {/* Pulse ring para estados com vendedores */}
                    {hasVendedores && !isSelected && (
                      <circle
                        cx={pos.cx}
                        cy={pos.cy}
                        r={pos.r + 4}

                      cx={pos.cx}

              animate={{ opacity: 1, y: 0, height: "auto" }}
              exit={{ opacity: 0, y: 20, height: 0 }}
              transition={{ duration: 0.35, ease: "easeOut" }}
              className="overflow-hidden"

                    </div>
                    <div>
                      <h3 className="text-base md:text-lg font-bold text-gray-900">{selectedEstado.nome}</h3>
                      <p className="text-xs text-gray-500">{selectedEstado.regiao}</p>
                    </div>
                  </div>
                  <div className="flex items-center gap-1.5 text-xs text-gray-500">
                    <Users className="w-3.5 h-3.5" />
                    <span>{selectedEstado.vendedores.length} vendedor{selectedEstado.vendedores.length > 1 ? "es" : ""}</span>
                  </div>
                </div>

                <div className="space-y-3">
                  {selectedEstado.vendedores.map((v, i) => (
                    <motion.div
                      key={i}
                      initial={{ opacity: 0, x: -10 }}
                      animate={{ opacity: 1, x: 0 }}
                      transition={{ delay: i * 0.05 }}
                      className="flex flex-col sm:flex-row sm:items-center gap-2 sm:gap-4 p-3 rounded-lg bg-gray-50 hover:bg-regina-red/5 transition-colors border border-gray-100 group"
                    >
                      <div className="flex-1">
                        <p className="text-sm font-semibold text-gray-900 group-hover:text-regina-red transition-colors">{v.nome}</p>

                          <Building2 className="w-3 h-3 text-gray-400" />
                          <span className="text-xs text-gray-500">{v.empresa}</span>
                        </div>
                      </div>

                        <a
                          href={`tel:${v.telefone.replace(/\D/g, "")}`}
                          className="flex items-center gap-1 text-xs text-gray-600 hover:text-regina-red transition-colors"
                        >
                          <Phone className="w-3 h-3" />
                          {v.telefone}
                        </a>
                        <a
                          href={`mailto:${v.email}`}
                          className="flex items-center gap-1 text-xs text-gray-600 hover:text-regina-red transition-colors"
                        >
                          <Mail className="w-3 h-3" />
                          <span className="hidden sm:inline">{v.email}</span>
                        </a>
                      </div>
                    </motion.div>
                  ))}
                </div>
              </div>
            </motion.div>
          )}
        </AnimatePresence>
      </div>
    </section>
  );
}
