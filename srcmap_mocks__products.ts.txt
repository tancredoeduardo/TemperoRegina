export function findProductById(id: string): { product: Product; category: Category } | null {
  for (const category of productCategories) {
    const product = category.products.find((p) => p.id === id);
    if (product) {
      return { product, category };
    }
  }
  return null;
}

export function getAllProducts(): (Product & { categoryId: string; categoryTitle: string })[] {
  const all: (Product & { categoryId: string; categoryTitle: string })[] = [];
  for (const category of productCategories) {
    for (const product of category.products) {
      all.push({ ...product, categoryId: category.id, categoryTitle: category.title });
    }
  }
  return all;
}
