/* ════════════════════════════════════════
   Skeleton loader
   ════════════════════════════════════════ */
function SkeletonPulse({ className = "" }: { className?: string }) {
  return (
    <div className={`bg-gray-200 rounded-md animate-pulse ${className}`} />
  );
}

function SkeletonCard({ lines = 3 }: { lines?: number }) {
  return (
    <div className="bg-white rounded-xl p-5 md:p-6 border border-gray-100 space-y-3">
      <SkeletonPulse className="h-5 w-20" />
      <SkeletonPulse className="h-6 w-3/4" />
      {Array.from({ length: lines }).map((_, i) => (
        <SkeletonPulse key={i} className="h-3 w-full" />
      ))}
    </div>
  );
}

/* ════════════════════════════════════════
   Timeline Item
   ════════════════════════════════════════ */
function TimelineItem({
  event,
  index,
}: {
  event: (typeof timelineData)[0];
  index: number;
}) {
  const ref = useRef<HTMLDivElement>(null);
  const isInView = useInView(ref, { once: true, margin: "-60px" });
  const isLeft = index % 2 === 0;

  return (
    <div
      ref={ref}
      className={`relative flex flex-col md:flex-row items-center md:items-start gap-4 md:gap-0 mb-10 md:mb-14 ${
        isLeft ? "md:flex-row" : "md:flex-row-reverse"
      }`}
    >
      {/* Content card */}
      <motion.div
        className="flex-1"
        initial={{
          opacity: 0,
          x: isLeft ? -60 : 60,
          scale: 0.95,
        }}
        animate={
          isInView
            ? { opacity: 1, x: 0, scale: 1 }
            : { opacity: 0, x: isLeft ? -60 : 60, scale: 0.95 }
        }
        transition={{ duration: 0.7, ease: [0.25, 0.46, 0.45, 0.94] }}
      >
        <div

            isLeft ? "md:text-right md:mr-8" : "md:text-left md:ml-8"

        >
          <motion.span
            className="inline-block bg-regina-red text-white text-xs font-bold px-3 py-1 rounded-full mb-3"
            whileHover={{ scale: 1.08 }}
            transition={{ type: "spring", stiffness: 400 }}
          >
            {event.year}
          </motion.span>
          <h3 className="text-base md:text-lg font-bold text-gray-900 mb-2 group-hover:text-regina-red transition-colors duration-300">
            {event.title}
          </h3>

          </p>

              className="mt-3 rounded-lg overflow-hidden h-32 md:h-40"

              />

        />

    </div>

}

/* ════════════════════════════════════════
   Animated counter
   ════════════════════════════════════════ */
function AnimatedCounter({
  target,
  suffix = "",
}: {
  target: number;
  suffix?: string;
}) {
  const ref = useRef<HTMLSpanElement>(null);
  const isInView = useInView(ref, { once: true });
  const [count, setCount] = useState(0);

  useEffect(() => {
    if (!isInView) return;
    let start = 0;
    const duration = 2000;
    const stepTime = Math.max(duration / target, 16);
    const timer = setInterval(() => {
      start += Math.ceil(target / (duration / stepTime));
      if (start >= target) {
        setCount(target);
        clearInterval(timer);
      } else {
        setCount(start);
      }
    }, stepTime);
    return () => clearInterval(timer);
  }, [isInView, target]);

  return (
    <span ref={ref}>
      {count}
      {suffix}
    </span>
  );
}

      </motion.div>
