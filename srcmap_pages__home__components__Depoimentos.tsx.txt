import { useState } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { ChevronLeft, ChevronRight, Quote, Star } from "lucide-react";
import { testimonialsData } from "@/mocks/testimonials";
import ScrollReveal from "@/components/base/ScrollReveal";

const cardVariants = {
  enter: (direction: number) => ({
    x: direction > 0 ? 200 : -200,
    opacity: 0,
    scale: 0.92,
  }),
  center: {
    x: 0,
    opacity: 1,
    scale: 1,
    transition: { duration: 0.5, ease: [0.32, 0.72, 0, 1] },
  },
  exit: (direction: number) => ({
    x: direction > 0 ? -200 : 200,
    opacity: 0,
    scale: 0.92,
    transition: { duration: 0.4, ease: [0.32, 0.72, 0, 1] },
  }),
};

const staggerContainer = {
  hidden: { opacity: 0 },
  visible: {
    opacity: 1,
    transition: { staggerChildren: 0.1, delayChildren: 0.15 },
  },
};

const staggerItem = {
  hidden: { opacity: 0, y: 20 },
  visible: {
    opacity: 1,
    y: 0,
    transition: { duration: 0.5, ease: [0.25, 0.46, 0.45, 0.94] },
  },
};

export default function Depoimentos() {
  const [current, setCurrent] = useState(0);
  const [direction, setDirection] = useState(0);

  // Sempre mostra 3 por página no desktop, 1 no mobile
  const itemsPerPage = typeof window !== "undefined" && window.innerWidth < 768 ? 1 : 3;
  const totalPages = Math.ceil(testimonialsData.length / itemsPerPage);

  const paginate = (newDirection: number) => {
    setDirection(newDirection);
    setCurrent((prev) => {
      const next = prev + newDirection;
      if (next < 0) return totalPages - 1;
      if (next >= totalPages) return 0;
      return next;
    });
  };

  const goTo = (index: number) => {
    setDirection(index > current ? 1 : -1);
    setCurrent(index);
  };

  const visibleTestimonials = testimonialsData.slice(
    current * itemsPerPage,
    current * itemsPerPage + itemsPerPage
  );

  return (
    <section
      id="depoimentos"
      className="relative bg-regina-cream py-14 md:py-24 overflow-hidden"
    >
      {/* Decorative */}
      <div className="absolute top-10 left-10 w-32 h-32 rounded-full bg-regina-red/[0.03]" />
      <div className="absolute bottom-10 right-10 w-40 h-40 rounded-full bg-regina-yellow/[0.05]" />

      <div className="relative max-w-6xl mx-auto px-4 md:px-6">
        <ScrollReveal>
          <div className="text-center mb-10 md:mb-14">
            <motion.div
              className="inline-flex items-center gap-2 mb-3"
              initial={{ opacity: 0, scale: 0.8 }}
              whileInView={{ opacity: 1, scale: 1 }}
              viewport={{ once: true }}
              transition={{ duration: 0.4 }}
            >
              <Quote className="w-4 h-4 text-regina-red" />
              <span className="text-xs font-semibold text-regina-red uppercase tracking-wider">
                Depoimentos
              </span>
            </motion.div>
            <h2 className="text-xl md:text-3xl font-bold text-gray-900 uppercase tracking-wide">
              O que Dizem Sobre Nós
            </h2>
            <p className="text-xs md:text-sm text-gray-500 mt-3 max-w-lg mx-auto">
              Quem usa Regina, aprova. Confira o que nossos parceiros e clientes dizem.
            </p>
          </div>
        </ScrollReveal>

        <ScrollReveal delay={0.1}>
          <div className="relative">
            <AnimatePresence mode="wait" custom={direction}>
              <motion.div
                key={current}
                custom={direction}
                variants={staggerContainer}
                initial="hidden"
                animate="visible"
                exit="hidden"
                className="grid grid-cols-1 md:grid-cols-3 gap-5 md:gap-6"
              >
                {visibleTestimonials.map((t) => (
                  <motion.div
                    key={t.id}
                    variants={staggerItem}
                    className="group bg-white rounded-xl p-5 md:p-6 border border-gray-100 hover:border-regina-red/20 hover:shadow-xl hover:-translate-y-1 transition-all duration-300"
                  >
                    {/* Stars */}
                    <div className="flex items-center gap-0.5 mb-3">
                      {[...Array(5)].map((_, i) => (
                        <Star key={i} className="w-3.5 h-3.5 text-regina-yellow fill-regina-yellow" />
                      ))}
                    </div>

                    <Quote className="w-6 h-6 text-regina-red/15 mb-3" />
                    <p className="text-sm text-gray-700 leading-relaxed mb-5 italic">
                      &ldquo;{t.text}&rdquo;
                    </p>
                    <div className="flex items-center gap-3 pt-3 border-t border-gray-100">
                      <motion.div
                        whileHover={{ scale: 1.1 }}
                        className={`w-10 h-10 md:w-11 md:h-11 rounded-full ${t.bgColor} flex items-center justify-center text-white text-sm font-bold shrink-0`}
                      >
                        {t.initials}
                      </motion.div>
                      <div>
                        <p className="text-sm font-bold text-gray-900">{t.name}</p>
                        <p className="text-[11px] text-gray-500">
                          {t.role}, {t.company}
                        </p>
                      </div>
                    </div>
                  </motion.div>
                ))}
              </motion.div>
            </AnimatePresence>

            {/* Navigation */}
            <div className="flex items-center justify-center gap-3 mt-8 md:mt-10">
              <motion.button
                type="button"
                onClick={() => paginate(-1)}
                whileHover={{ scale: 1.1 }}
                whileTap={{ scale: 0.95 }}
                className="w-9 h-9 flex items-center justify-center rounded-full bg-white border border-gray-200 text-regina-red hover:bg-regina-red hover:text-white hover:border-regina-red transition-all"
                aria-label="Anterior"
              >
                <ChevronLeft className="w-4 h-4" />
              </motion.button>
              {Array.from({ length: totalPages }).map((_, idx) => (
                <button
                  key={idx}
                  type="button"
                  onClick={() => goTo(idx)}
                  className="relative h-2 rounded-full overflow-hidden bg-gray-300 transition-all duration-400"
                  style={{ width: idx === current ? 28 : 8 }}
                  aria-label={`Página ${idx + 1}`}
                >
                  {idx === current && (
                    <motion.div
                      className="absolute inset-0 bg-regina-red"
                      layoutId="testimonial-dot"
                      transition={{ type: "spring", stiffness: 400, damping: 30 }}
                    />
                  )}
                </button>
              ))}
              <motion.button
                type="button"
                onClick={() => paginate(1)}
                whileHover={{ scale: 1.1 }}
                whileTap={{ scale: 0.95 }}
                className="w-9 h-9 flex items-center justify-center rounded-full bg-white border border-gray-200 text-regina-red hover:bg-regina-red hover:text-white hover:border-regina-red transition-all"
                aria-label="Próximo"
              >
                <ChevronRight className="w-4 h-4" />
              </motion.button>
            </div>
          </div>
        </ScrollReveal>
      </div>
    </section>
  );
}
