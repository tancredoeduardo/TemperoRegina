import { useState, useEffect } from "react";
import { Link } from "react-router-dom";
import {
  Search,
  ChevronRight,
  Calendar,
  Clock,
  ArrowLeft,
  ArrowRight,
  Tag,
} from "lucide-react";
import Navbar from "@/components/feature/Navbar";
import Footer from "@/components/feature/Footer";
import ScrollReveal from "@/components/base/ScrollReveal";
import { blogPosts, blogCategories } from "@/mocks/blog";

export default function BlogPage() {
  const [activeCategory, setActiveCategory] = useState("Todas");
  const [searchTerm, setSearchTerm] = useState("");
  const [selectedPost, setSelectedPost] = useState<number | null>(null);
  const [heroVisible, setHeroVisible] = useState(false);
  const [currentPage, setCurrentPage] = useState(1);

  useEffect(() => {
    setHeroVisible(true);
  }, []);

  useEffect(() => {
    if (selectedPost !== null) {
      document.body.style.overflow = "hidden";
    } else {
      document.body.style.overflow = "";
    }
    return () => {
      document.body.style.overflow = "";
    };
  }, [selectedPost]);

  const filtered = blogPosts.filter((post) => {
    const matchesCategory = activeCategory === "Todas" || post.category === activeCategory;
    const matchesSearch =
      post.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
      post.excerpt.toLowerCase().includes(searchTerm.toLowerCase());
    return matchesCategory && matchesSearch;
  });

  const postsPerPage = 6;
  const totalPages = Math.ceil(filtered.length / postsPerPage);
  const paginatedPosts = filtered.slice((currentPage - 1) * postsPerPage, currentPage * postsPerPage);

  useEffect(() => {
    setCurrentPage(1);
  }, [activeCategory, searchTerm]);

  const activePost = selectedPost ? blogPosts.find((p) => p.id === selectedPost) : null;

  const formatDate = (dateStr: string) => {
    const date = new Date(dateStr);
    return date.toLocaleDateString("pt-BR", { day: "numeric", month: "long", year: "numeric" });
  };

  return (
    <div className="min-h-screen bg-regina-gray-warm flex flex-col">
      <Navbar />

      <div className="relative w-full h-[240px] md:h-[340px] overflow-hidden" style={{ marginTop: "80px" }}>
        <img
          src="https://readdy.ai/api/search-image?query=A vibrant Brazilian kitchen scene with fresh colorful ingredients, herbs and spices spread across a rustic wooden table, warm golden natural lighting streaming through a window, professional food photography, appetizing and inviting atmosphere with red and yellow accents, editorial magazine style"
          alt="Blog Tempero Regina"
          className="w-full h-full object-cover"
        />
        <div className="absolute inset-0 bg-gradient-to-b from-black/40 via-black/30 to-black/50" />
        <div className="absolute inset-0 flex items-center justify-center">
          <div className="text-center px-4">
            <h1 className={`text-3xl md:text-5xl font-bold text-white uppercase tracking-wide mb-3 drop-shadow-lg transition-all duration-700 ${heroVisible ? "opacity-100 translate-y-0" : "opacity-0 translate-y-6"}`}>
              Notícias e Blog
            </h1>
            <p className={`text-sm md:text-lg text-white/90 max-w-xl mx-auto drop-shadow-md transition-all duration-700 delay-150 ${heroVisible ? "opacity-100 translate-y-0" : "opacity-0 translate-y-6"}`}>
              Fique por dentro das novidades, lançamentos e histórias da Tempero Regina
            </p>
          </div>
        </div>
      </div>

      <section className="py-5 md:py-6 bg-white border-b border-gray-100 sticky top-16 md:top-20 z-30">
        <div className="max-w-6xl mx-auto px-4 md:px-6">
          <div className="flex flex-col md:flex-row gap-4 md:gap-6 items-start md:items-center justify-between">
            <div className="relative w-full md:w-80">
              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
              <input type="text" placeholder="Buscar notícias..." value={searchTerm} onChange={(e) => setSearchTerm(e.target.value)} className="w-full pl-9 pr-4 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:border-regina-red focus:ring-1 focus:ring-regina-red/20 bg-white" />
            </div>
            <div className="flex items-center gap-2 overflow-x-auto scrollbar-hide w-full md:w-auto">
              {blogCategories.map((cat) => (
                <button key={cat} type="button" onClick={() => setActiveCategory(cat)} className={`px-3 py-1.5 text-xs md:text-sm font-medium rounded-full whitespace-nowrap transition-all ${activeCategory === cat ? "bg-regina-red text-white shadow-sm" : "bg-gray-100 text-gray-600 hover:bg-gray-200"}`}>
                  {cat}
                </button>
              ))}
            </div>
          </div>
        </div>
      </section>

      <section className="flex-1 py-8 md:py-12">
        <div className="max-w-6xl mx-auto px-4 md:px-6">
          <nav className="flex items-center gap-2 text-xs md:text-sm text-gray-500 mb-6">
            <Link to="/" className="hover:text-regina-red transition-colors">Home</Link>
            <ChevronRight className="w-3 h-3" />
            <span className="text-gray-800 font-medium">Blog</span>
            {activeCategory !== "Todas" && <><ChevronRight className="w-3 h-3" /><span className="text-regina-red font-medium">{activeCategory}</span></>}
          </nav>

          <ScrollReveal>
            {filtered.length === 0 ? (
              <div className="text-center py-16">
                <Search className="w-12 h-12 text-gray-300 mx-auto mb-3" />
                <p className="text-gray-500 text-sm md:text-base">Nenhuma notícia encontrada.</p>
                <button type="button" onClick={() => { setSearchTerm(""); setActiveCategory("Todas"); }} className="mt-3 text-sm text-regina-red hover:underline font-medium">Limpar filtros</button>
              </div>
            ) : (
              <>
                <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-5 md:gap-6">
                  {paginatedPosts.map((post, idx) => (
                    <article key={post.id} className="group bg-white rounded-xl overflow-hidden border border-gray-100 hover:shadow-xl hover:-translate-y-1 hover:border-regina-red/20 transition-all duration-300 cursor-pointer flex flex-col" onClick={() => setSelectedPost(post.id)} style={{ animationDelay: `${idx * 50}ms` }}>
                      <div className="relative h-44 md:h-52 overflow-hidden">
                        <img src={post.image} alt={post.title} className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500" />
                        <div className="absolute inset-0 bg-gradient-to-t from-black/40 to-transparent" />
                        <span className="absolute top-3 left-3 bg-regina-red text-white text-[10px] md:text-xs font-bold px-2 py-1 rounded-full uppercase tracking-wide">{post.category}</span>
                      </div>
                      <div className="p-4 md:p-5 flex flex-col flex-1">
                        <div className="flex items-center gap-3 text-xs text-gray-400 mb-2">
                          <div className="flex items-center gap-1"><Calendar className="w-3 h-3" /><span>{formatDate(post.date)}</span></div>
                          <div className="flex items-center gap-1"><Clock className="w-3 h-3" /><span>{post.readTime} de leitura</span></div>
                        </div>
                        <h3 className="text-sm md:text-base font-bold text-gray-900 mb-2 line-clamp-2 group-hover:text-regina-red transition-colors duration-300 flex-1">{post.title}</h3>
                        <p className="text-xs md:text-sm text-gray-500 line-clamp-2 mb-3">{post.excerpt}</p>
                        <div className="flex items-center gap-1 text-xs font-medium text-regina-red group-hover:gap-2 transition-all mt-auto"><span>Ler mais</span><ArrowRight className="w-3 h-3" /></div>
                      </div>
                    </article>
                  ))}
                </div>

                {totalPages > 1 && (
                  <div className="flex items-center justify-center gap-2 mt-10">
                    <button type="button" onClick={() => setCurrentPage((p) => Math.max(1, p - 1))} disabled={currentPage === 1} className="w-8 h-8 flex items-center justify-center rounded-full border border-gray-200 text-gray-600 hover:border-regina-red hover:text-regina-red transition-all disabled:opacity-40 disabled:cursor-not-allowed"><ArrowLeft className="w-4 h-4" /></button>
                    {Array.from({ length: totalPages }).map((_, idx) => (
                      <button key={idx} type="button" onClick={() => setCurrentPage(idx + 1)} className={`w-8 h-8 flex items-center justify-center rounded-full text-sm font-medium transition-all ${currentPage === idx + 1 ? "bg-regina-red text-white" : "border border-gray-200 text-gray-600 hover:border-regina-red hover:text-regina-red"}`}>{idx + 1}</button>
                    ))}
                    <button type="button" onClick={() => setCurrentPage((p) => Math.min(totalPages, p + 1))} disabled={currentPage === totalPages} className="w-8 h-8 flex items-center justify-center rounded-full border border-gray-200 text-gray-600 hover:border-regina-red hover:text-regina-red transition-all disabled:opacity-40 disabled:cursor-not-allowed"><ArrowRight className="w-4 h-4" /></button>
                  </div>
                )}
              </>
            )}
          </ScrollReveal>
        </div>
      </section>

      <ScrollReveal>
        <section className="py-10 md:py-14 bg-regina-red">
          <div className="max-w-4xl mx-auto px-4 md:px-6 text-center">
            <h2 className="text-xl md:text-2xl font-bold text-white uppercase tracking-wide mb-3">Quer receber nossas novidades?</h2>
            <p className="text-sm md:text-base text-white/80 mb-6 max-w-lg mx-auto">Cadastre seu e-mail e fique por dentro de lançamentos, receitas e eventos da Tempero Regina.</p>
            <form method="POST" action="https://readdy.ai/api/form/d7qmhmsdj3knvn2p290g" data-readdy-form className="flex flex-col sm:flex-row items-center gap-3 max-w-md mx-auto" onSubmit={(e) => { const email = (e.currentTarget.elements.namedItem("email") as HTMLInputElement)?.value; if (!email || !email.includes("@")) { e.preventDefault(); alert("Por favor, insira um e-mail válido."); } }}>
              <input type="email" name="email" placeholder="seu@email.com" required className="w-full sm:flex-1 px-4 py-2.5 rounded-lg text-sm text-gray-900 placeholder-gray-400 focus:outline-none focus:ring-2 focus:ring-regina-yellow" />
              <button type="submit" className="w-full sm:w-auto bg-regina-yellow text-regina-dark text-sm font-bold px-6 py-2.5 rounded-lg hover:bg-regina-yellow-light transition-all duration-300 whitespace-nowrap">Cadastrar</button>
            </form>
          </div>
        </section>
      </ScrollReveal>

      {activePost && (
        <div className="fixed inset-0 z-[60] bg-black/50 backdrop-blur-sm flex items-start justify-center px-4 py-8 md:py-12 overflow-y-auto" onClick={() => setSelectedPost(null)}>
          <div className="bg-white rounded-2xl max-w-3xl w-full my-auto animate-scale-in" onClick={(e) => e.stopPropagation()}>
            <div className="relative h-48 md:h-72 overflow-hidden rounded-t-2xl">
              <img src={activePost.image} alt={activePost.title} className="w-full h-full object-cover" />
              <div className="absolute inset-0 bg-gradient-to-t from-black/60 to-transparent" />
              <button type="button" onClick={() => setSelectedPost(null)} className="absolute top-4 right-4 w-9 h-9 flex items-center justify-center rounded-full bg-white/90 hover:bg-white text-gray-700 transition-all" aria-label="Fechar"><ArrowLeft className="w-4 h-4" /></button>
              <div className="absolute bottom-4 left-4 right-4">
                <div className="flex items-center gap-2 mb-2"><span className="bg-regina-red text-white text-xs font-bold px-2 py-0.5 rounded-full uppercase">{activePost.category}</span></div>
                <h2 className="text-xl md:text-3xl font-bold text-white drop-shadow-lg">{activePost.title}</h2>
              </div>
            </div>
            <div className="p-5 md:p-8">
              <div className="flex items-center gap-4 text-xs md:text-sm text-gray-500 mb-6 pb-6 border-b border-gray-100">
                <div className="flex items-center gap-1.5"><Calendar className="w-4 h-4 text-regina-red" /><span>{formatDate(activePost.date)}</span></div>
                <div className="flex items-center gap-1.5"><Clock className="w-4 h-4 text-regina-red" /><span>{activePost.readTime} de leitura</span></div>
                <div className="flex items-center gap-1.5"><Tag className="w-4 h-4 text-regina-red" /><span>{activePost.category}</span></div>
              </div>
              <div className="prose prose-sm md:prose-base max-w-none text-gray-700 leading-relaxed whitespace-pre-line">{activePost.content}</div>
              <div className="mt-8 pt-6 border-t border-gray-100">
                <button type="button" onClick={() => setSelectedPost(null)} className="inline-flex items-center gap-2 text-sm font-medium text-regina-red hover:text-regina-red-dark transition-colors"><ArrowLeft className="w-4 h-4" />Voltar para o Blog</button>
              </div>
            </div>
          </div>
        </div>
      )}

      <Footer />
    </div>
  );
}
